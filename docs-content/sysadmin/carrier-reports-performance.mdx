### On-Time Delivery Calculation

<CodeBlock language="typescript">
{`export async function getOnTimeDeliveryRate(
  dateRange: DateRangeSchema
): Promise<ActionState<number>> {
  await requireRole([UserRole.OWNER, UserRole.MANAGER]);
  const tenantId = await requireTenantId();
  const prisma = await getTenantPrisma();

  const loads = await prisma.load.findMany({
    where: {
      tenantId,
      status: 'DELIVERED',
      deliveredAt: {
        gte: dateRange.startDate,
        lte: dateRange.endDate,
      },
    },
    include: { deliveryStop: true },
  });

  const onTimeCount = loads.filter(load =>
    load.deliveredAt &&
    load.deliveryStop?.scheduledTime &&
    load.deliveredAt <= load.deliveryStop.scheduledTime
  ).length;

  const rate = loads.length > 0 ? (onTimeCount / loads.length) * 100 : 0;

  return { success: true, data: rate };
}`}
</CodeBlock>
