### Status Computation Logic

<CodeBlock language="typescript">
{`function computeTruckStatus(lastReport: GpsReport): TruckStatus {
  const now = new Date();
  const minutesSinceUpdate = differenceInMinutes(now, lastReport.timestamp);

  // Offline: no GPS in 10+ minutes
  if (minutesSinceUpdate > 10) return 'OFFLINE';

  // Idle: stopped but recent GPS
  if (lastReport.speed < 5) return 'IDLE';

  // Moving: speed >= 5 mph
  return 'MOVING';
}`}
</CodeBlock>
