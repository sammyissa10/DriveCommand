### Trigger Handler

<CodeBlock language="typescript">
{`// Called when a load is dispatched
export async function onLoadDispatched(load: Load) {
  const rules = await prisma.autoStartRule.findMany({
    where: {
      tenantId: load.tenantId,
      triggerEvent: 'LOAD_DISPATCHED',
      isActive: true,
    },
  });

  for (const rule of rules) {
    // Evaluate conditions
    if (rule.conditions && !evaluateConditions(rule.conditions, load)) {
      continue;
    }

    // Start the workflow
    await prisma.activeChecklist.create({
      data: {
        tenantId: load.tenantId,
        checklistId: rule.checklistId,
        assigneeId: load.driverId,
        linkedEntityId: load.id,
        linkedEntityType: 'LOAD',
      },
    });

    // Send notification to driver
    await sendPushToUser(load.driverId, {
      title: 'New Checklist',
      body: \`Complete "\${rule.checklist.name}" before starting this load\`,
    });
  }
}`}
</CodeBlock>
