### `sendMessage()`

Sends a targeted message to a specific driver with push notification.

<CodeBlock language="typescript">
{`export async function sendMessage(
  prevState: ActionState | null,
  formData: FormData
): Promise<ActionState<FleetMessage>> {
  await requireRole([UserRole.OWNER, UserRole.MANAGER]);
  const tenantId = await requireTenantId();
  const prisma = await getTenantPrisma();

  const parsed = SendMessageSchema.parse({
    recipientId: formData.get('recipientId'),
    message: formData.get('message'),
    loadId: formData.get('loadId'), // optional
  });

  const fleetMessage = await prisma.fleetMessage.create({
    data: {
      tenantId,
      senderId: (await getCurrentUser()).id,
      recipientId: parsed.recipientId,
      message: parsed.message,
      loadId: parsed.loadId,
      isBroadcast: false,
    },
  });

  // Send push notification
  await sendPushToUser(parsed.recipientId, {
    title: 'New Message',
    body: parsed.message.substring(0, 100),
    data: { type: 'fleet_message', messageId: fleetMessage.id },
  });

  return { success: true, data: fleetMessage };
}`}
</CodeBlock>
