### Example Aggregation Query

<CodeBlock language="typescript">
{`export async function getRevenueByCustomer(
  dateRange: DateRangeSchema
): Promise<ActionState<RevenueByCustomer[]>> {
  await requireRole([UserRole.OWNER, UserRole.MANAGER]);
  const tenantId = await requireTenantId();
  const prisma = await getTenantPrisma();

  const results = await prisma.invoice.groupBy({
    by: ['customerId'],
    where: {
      tenantId,
      createdAt: {
        gte: dateRange.startDate,
        lte: dateRange.endDate,
      },
    },
    _sum: { total: true },
    _count: { id: true },
  });

  // Enrich with customer names
  const enriched = await Promise.all(
    results.map(async (r) => {
      const customer = await prisma.customer.findUnique({
        where: { id: r.customerId },
      });
      return {
        customerId: r.customerId,
        customerName: customer?.companyName ?? 'Unknown',
        totalRevenue: r._sum.total ?? 0,
        invoiceCount: r._count.id,
      };
    })
  );

  return { success: true, data: enriched };
}`}
</CodeBlock>
