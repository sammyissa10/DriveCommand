### Lane Profit Calculation

<CodeBlock language="typescript">
{`async function calculateLaneProfitability(
  origin: string,
  destination: string,
  dateRange: DateRange
): Promise<LaneMetrics> {
  const routes = await prisma.route.findMany({
    where: {
      tenantId,
      origin,
      destination,
      createdAt: { gte: dateRange.start, lte: dateRange.end },
    },
    include: { loads: { include: { invoice: true } } },
  });

  // Aggregate revenue from invoices
  const revenue = routes.reduce((sum, route) =>
    sum + route.loads.reduce((s, l) => s + (l.invoice?.total ?? 0), 0), 0
  );

  // Aggregate fuel costs
  const fuelCost = await prisma.fuelLog.aggregate({
    where: { routeId: { in: routes.map(r => r.id) } },
    _sum: { totalCost: true },
  });

  // Aggregate driver pay
  const driverPay = await prisma.payrollRecord.aggregate({
    where: { routeId: { in: routes.map(r => r.id) } },
    _sum: { basePay: true },
  });

  const expenses = (fuelCost._sum.totalCost ?? 0) + (driverPay._sum.basePay ?? 0);
  const profit = revenue - expenses;
  const margin = revenue > 0 ? (profit / revenue) * 100 : 0;

  return { revenue, expenses, profit, margin, loadCount: routes.length };
}`}
</CodeBlock>
