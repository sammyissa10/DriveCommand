### Task Completion Validation

<CodeBlock language="typescript">
{`export async function completeTask(
  prevState: ActionState | null,
  formData: FormData
): Promise<ActionState<Task>> {
  const parsed = CompleteTaskSchema.parse({
    taskId: formData.get('taskId'),
    stepType: formData.get('stepType'),
    value: formData.get('value'),
    photoUrl: formData.get('photoUrl'),
  });

  const task = await prisma.task.findUnique({
    where: { id: parsed.taskId },
    include: { step: true },
  });

  // Validate based on step type
  if (task.step.type === 'PHOTO_REQUIRED' && !parsed.photoUrl) {
    return { success: false, error: 'Photo required' };
  }

  if (task.step.type === 'TEXT_ENTRY' && parsed.value.length < 10) {
    return { success: false, error: 'Text must be at least 10 characters' };
  }

  const updated = await prisma.task.update({
    where: { id: parsed.taskId },
    data: {
      completedAt: new Date(),
      value: parsed.value,
      photoUrl: parsed.photoUrl,
      duration: differenceInSeconds(new Date(), task.createdAt),
    },
  });

  return { success: true, data: updated };
}`}
</CodeBlock>
