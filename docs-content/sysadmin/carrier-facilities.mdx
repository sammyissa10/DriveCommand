### Geofence Check Logic

GPS reports trigger geofence checks on every position update:

<CodeBlock language="typescript">
{`async function checkGeofenceEntry(gpsReport: GpsReport) {
  const facilities = await prisma.facility.findMany({
    where: { tenantId: gpsReport.tenantId },
  });

  for (const facility of facilities) {
    const distance = calculateDistance(
      gpsReport.latitude,
      gpsReport.longitude,
      facility.latitude,
      facility.longitude
    );

    if (distance <= facility.radiusMeters) {
      await logCheckinEvent({
        facilityId: facility.id,
        truckId: gpsReport.truckId,
        timestamp: gpsReport.timestamp,
      });
    }
  }
}`}
</CodeBlock>
