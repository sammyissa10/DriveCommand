### Auto-Status Badge Logic

Truck status badges are computed server-side from multiple sources:

<CodeBlock language="typescript">
{`function computeTruckStatus(truck: Truck): TruckStatus {
  // Priority 1: Check if actively in use
  if (truck.currentRouteId) return 'IN_USE';

  // Priority 2: Check for maintenance
  const activeMaintenance = truck.scheduledServices.find(
    s => s.status === 'IN_PROGRESS'
  );
  if (activeMaintenance) return 'IN_MAINTENANCE';

  // Priority 3: Check for expired docs
  const expiredDocs = truck.documents.filter(
    d => d.expiryDate && d.expiryDate < new Date()
  );
  if (expiredDocs.length > 0) return 'EXPIRED_DOCS';

  // Default: Ready to use
  return 'READY_TO_USE';
}`}
</CodeBlock>
