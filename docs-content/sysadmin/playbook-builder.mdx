### Playbook JSON Schema

<CodeBlock language="typescript">
{`interface PlaybookDefinition {
  phases: Phase[];
  metadata: {
    name: string;
    version: number;
    createdBy: string;
  };
}

interface Phase {
  id: string;
  name: string;
  steps: Step[];
  conditions?: Condition[];
}

interface Step {
  id: string;
  name: string;
  type: 'checkbox' | 'photo' | 'signature' | 'text' | 'multiple_choice';
  required: boolean;
  instructions: string;
  conditions?: Condition[];
}

interface Condition {
  type: 'show_if' | 'branch_if';
  field: string; // Reference to previous step ID
  operator: 'equals' | 'not_equals' | 'contains';
  value: string;
  targetPhaseId?: string; // For branch_if
}`}
</CodeBlock>
