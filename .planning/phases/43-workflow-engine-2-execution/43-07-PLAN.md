**Test 1: Snapshot Immutability**

The spec says: "generate instance, mutate source Playbook, assert snapshot unchanged."

Since `generatePlaybookInstance` calls prisma, we mock prisma and control the playbook data. After calling `generatePlaybookInstance` with a playbook that has certain step data, we verify that the `playbookSnapshot` written to `prisma.playbookInstance.create` is a deep copy and not a reference to the original playbook object.

```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest'

// Mock prisma — we control what findFirst/create return
vi.mock('@/lib/db/prisma', () => ({
  prisma: {
    playbook: {
      findFirst: vi.fn(),
    },
    playbookInstance: {
      findFirst: vi.fn(),
      create: vi.fn(),
      findUniqueOrThrow: vi.fn(),
    },
    stepInstance: {
      create: vi.fn(),
      findMany: vi.fn(),
      update: vi.fn(),
    },
    user: {
      findFirst: vi.fn(),
      findMany: vi.fn(),
    },
    playbookNotification: {
      create: vi.fn(),
    },
    $transaction: vi.fn(),
  },
  TX_OPTIONS: {},
}))

vi.mock('@/lib/notifications/send-push', () => ({
  sendPushToUser: vi.fn().mockResolvedValue(undefined),
}))

describe('generatePlaybookInstance — snapshot immutability', () => {
  it('playbookSnapshot is a deep copy — mutating the source Playbook does not change the snapshot', async () => {
    const { generatePlaybookInstance } = await import('@/server/services/workflows/generatePlaybookInstance')
    const { prisma } = await import('@/lib/db/prisma')
    
    const mockStep = {
      stepTemplateId: 'step-tpl-1',
      sequence: 0,
      playbookPhase: 'DAY_1',
      isRequired: true,
      isDispatchBlocker: true,
      dueDaysFromStart: null,
      dueBeforeDispatch: false,
      overrideConfig: null,
      stepTemplate: {
        name: 'Upload License',
        stepType: 'DOCUMENT_UPLOAD',
        assigneeRole: 'ADMIN',
        formSchema: null,
        documentTypeName: "Driver's License",
        requiresPhoto: false,
        requiresSignature: false,
        description: 'Upload a valid CDL',
      },
    }
    
    const sourceMutablePlaybook = {
      id: 'playbook-1',
      tenantId: 'tenant-1',
      name: 'CDL Onboarding',
      category: 'DRIVER_ONBOARDING',
      entityType: 'DRIVER',
      isActive: true,
      deletedAt: null,
      steps: [mockStep],
    }
    
    let capturedSnapshot: unknown = null
    
    vi.mocked(prisma.playbook.findFirst).mockResolvedValue(sourceMutablePlaybook as any)
    vi.mocked(prisma.playbookInstance.findFirst).mockResolvedValue(null) // no duplicate
    vi.mocked(prisma.user.findFirst).mockResolvedValue({ id: 'driver-1' } as any)
    vi.mocked(prisma.user.findMany).mockResolvedValue([]) // no admin users → no assignee
    vi.mocked(prisma.stepInstance.findMany).mockResolvedValue([])
    vi.mocked(prisma.playbookInstance.findUniqueOrThrow).mockResolvedValue({ id: 'instance-1', stepInstances: [] } as any)
    
    // Capture the snapshot passed to create
    vi.mocked(prisma.$transaction).mockImplementation(async (fn: any) => {
      const txMock = {
        $executeRaw: vi.fn().mockResolvedValue(undefined),
        playbookInstance: {
          create: vi.fn().mockImplementation(({ data }: any) => {
            capturedSnapshot = JSON.parse(JSON.stringify(data.playbookSnapshot)) // deep clone what was saved
            return Promise.resolve({ id: 'instance-1', ...data })
          }),
        },
        stepInstance: { create: vi.fn().mockResolvedValue({}) },
      }
      return fn(txMock)
    })
    
    await generatePlaybookInstance({
      playbookId: 'playbook-1',
      entityType: 'DRIVER',
      entityId: 'driver-1',
      tenantId: 'tenant-1',
      triggeredBy: 'manual',
    })
    
    // Now mutate the source playbook
    sourceMutablePlaybook.name = 'MUTATED NAME'
    sourceMutablePlaybook.steps[0].stepTemplate.name = 'MUTATED STEP'
    
    // The captured snapshot must be unchanged
    const snap = capturedSnapshot as { name: string; steps: Array<{ stepTemplate: { name: string } }> }
    expect(snap.name).toBe('CDL Onboarding')
    expect(snap.steps[0].stepTemplate.name).toBe('Upload License')
  })
})
```

**`apps/mobile/src/__tests__/workflows-tap-targets.test.ts`:**

The spec mandates: "Mobile tap-target audit: test utility fails if any Pressable/TouchableOpacity has height < 56."

This test reads the StyleSheet from each task screen file and asserts height values. This is a static analysis test (no DOM rendering needed — no Expo test renderer required).

`apps/mobile` uses **jest-expo** (not vitest) — do NOT import from vitest. Jest globals (`describe`, `it`, `expect`) are available automatically via jest-expo's preset. Use `import * as fs` and `import * as path` normally (Node.js globals are available in jest).

```typescript
import * as fs from 'node:fs'
import * as path from 'node:path'

const SCREENS_DIR = path.resolve(__dirname, '../screens/workflows')
const MIN_TAP_TARGET_PX = 56

// Regex to find height values in StyleSheet definitions
// Matches: height: 48 or height: '48px' — must be ≥56
const HEIGHT_PATTERN = /height\s*:\s*(\d+)/g

// Files to audit
const AUDIT_FILES = [
  'MyTasksScreen.tsx',
  'DocumentUploadScreen.tsx',
  'FormFillScreen.tsx',
  'SignatureScreen.tsx',
]

describe('Mobile tap-target audit — workflow task screens', () => {
  AUDIT_FILES.forEach((fileName) => {
    it(`${fileName}: no Pressable/TouchableOpacity with height < ${MIN_TAP_TARGET_PX}px`, () => {
      const filePath = path.join(SCREENS_DIR, fileName)
      
      if (!fs.existsSync(filePath)) {
        // File doesn't exist yet — skip with a descriptive message
        console.warn(`SKIP: ${fileName} not found at ${filePath}`)
        return
      }
      
      const source = fs.readFileSync(filePath, 'utf-8')
      
      // Find all height values defined in StyleSheet.create({}) blocks
      // We look for height values inside style objects that are also near Pressable/TouchableOpacity
      const violations: number[] = []
      let match: RegExpExecArray | null
      
      // Reset lastIndex
      HEIGHT_PATTERN.lastIndex = 0
      
      while ((match = HEIGHT_PATTERN.exec(source)) !== null) {
        const heightValue = parseInt(match[1], 10)
        if (heightValue > 0 && heightValue < MIN_TAP_TARGET_PX) {
          // Check if this height is near a touchable element
          // Heuristic: look at 500 chars of context before this match
          const contextStart = Math.max(0, match.index - 500)
          const context = source.slice(contextStart, match.index + 20)
          const isTouchableContext =
            context.includes('TouchableOpacity') ||
            context.includes('Pressable') ||
            context.includes('button') ||
            context.includes('Button') ||
            context.includes('action')
          
          if (isTouchableContext) {
            violations.push(heightValue)
          }
        }
      }
      
      expect(violations).toHaveLength(0)
    })
  })
})
```
