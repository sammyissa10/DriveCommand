```typescript
// Fetch driver readiness — block if not ready AND no override
const driverUser = await prisma.user.findUnique({
  where: { id: linkedDriverUserId },
  select: { isDispatchReady: true, role: true },
});

if (driverUser && !driverUser.isDispatchReady && !args.overrideReason) {
  throw new Error('DRIVER_NOT_DISPATCH_READY'); // caller (API route) translates to 409 Conflict
}

// If override was provided, verify the current user is admin and log audit BEFORE creating dispatch
if (args.overrideReason) {
  const currentUser = await prisma.user.findUnique({
    where: { id: args.currentUserId },
    select: { role: true },
  });
  if (currentUser?.role !== 'owner' && currentUser?.role !== 'admin') {
    throw new Error('OVERRIDE_REQUIRES_ADMIN');
  }
}
```
