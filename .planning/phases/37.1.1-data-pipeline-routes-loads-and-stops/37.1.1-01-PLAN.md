4. **Append backfill SQL** after the schema changes. This backfill creates RouteStops for existing loads that have routeId set AND lat/lng populated:

```sql
-- Backfill PICKUP RouteStops for existing loads with routeId and coordinates
INSERT INTO "RouteStop" ("id", "routeId", "tenantId", "loadId", "type", "address", "lat", "lng", "position", "status", "geofenceHit", "createdAt", "updatedAt")
SELECT
  gen_random_uuid(),
  l."routeId",
  l."tenantId",
  l."id",
  'PICKUP'::"RouteStopType",
  l."origin",
  l."pickupLat",
  l."pickupLng",
  ROW_NUMBER() OVER (PARTITION BY l."routeId" ORDER BY l."pickupDate" ASC) * 2 - 1,
  'PENDING'::"RouteStopStatus",
  false,
  NOW(),
  NOW()
FROM "Load" l
WHERE l."routeId" IS NOT NULL
  AND l."pickupLat" IS NOT NULL
  AND l."archivedAt" IS NULL
  AND NOT EXISTS (
    SELECT 1 FROM "RouteStop" rs
    WHERE rs."loadId" = l."id" AND rs."type" = 'PICKUP'::"RouteStopType"
  );

-- Backfill DELIVERY RouteStops for existing loads with routeId and coordinates
INSERT INTO "RouteStop" ("id", "routeId", "tenantId", "loadId", "type", "address", "lat", "lng", "position", "status", "geofenceHit", "createdAt", "updatedAt")
SELECT
  gen_random_uuid(),
  l."routeId",
  l."tenantId",
  l."id",
  'DELIVERY'::"RouteStopType",
  l."destination",
  l."deliveryLat",
  l."deliveryLng",
  ROW_NUMBER() OVER (PARTITION BY l."routeId" ORDER BY l."deliveryDate" ASC) * 2,
  'PENDING'::"RouteStopStatus",
  false,
  NOW(),
  NOW()
FROM "Load" l
WHERE l."routeId" IS NOT NULL
  AND l."deliveryLat" IS NOT NULL
  AND l."archivedAt" IS NULL
  AND NOT EXISTS (
    SELECT 1 FROM "RouteStop" rs
    WHERE rs."loadId" = l."id" AND rs."type" = 'DELIVERY'::"RouteStopType"
  );

-- Backfill pickupStopId and deliveryStopId on Load from the RouteStops just created
UPDATE "Load" l
SET "pickupStopId" = rs."id"
FROM "RouteStop" rs
WHERE rs."loadId" = l."id" AND rs."type" = 'PICKUP'::"RouteStopType"
  AND l."pickupStopId" IS NULL;

UPDATE "Load" l
SET "deliveryStopId" = rs."id"
FROM "RouteStop" rs
WHERE rs."loadId" = l."id" AND rs."type" = 'DELIVERY'::"RouteStopType"
  AND l."deliveryStopId" IS NULL;
```
