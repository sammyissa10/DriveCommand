### 6. Extend api-client with driver endpoints
File: packages/api-client/src/driver.ts

```typescript
export const driverApi = {
  getDashboard: (token: string) => apiRequest<DriverDashboardData>('/api/mobile/driver/dashboard', { token }),
  getLoads: (token: string, status: 'active' | 'history') =>
    apiRequest<Load[]>(`/api/mobile/driver/loads?status=${status}`, { token }),
  getLoad: (token: string, id: string) =>
    apiRequest<LoadDetail>('/api/mobile/driver/loads/' + id, { token }),
  updateLoadStatus: (token: string, id: string, status: LoadStatus) =>
    apiRequest<{ load: Load }>('/api/mobile/driver/loads/' + id + '/status', {
      method: 'POST', token, body: JSON.stringify({ status }),
    }),
}
```

### 8. Create TanStack Query provider
Install: `npm install @tanstack/react-query`

File: apps/mobile/context/QueryProvider.tsx
```typescript
import { QueryClient, QueryClientProvider } from '@tanstack/react-query'
const queryClient = new QueryClient({
  defaultOptions: { queries: { staleTime: 30_000, retry: 2 } }
})
export function QueryProvider({ children }) {
  return <QueryClientProvider client={queryClient}>{children}</QueryClientProvider>
}
```

Add QueryProvider to root _layout.tsx (wraps AuthProvider).
