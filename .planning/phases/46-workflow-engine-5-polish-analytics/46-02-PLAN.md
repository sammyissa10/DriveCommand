    ```typescript
    /**
     * Twilio SMS helper — graceful no-op when env vars are absent.
     *
     * Never throws. Logs a warning and returns early if:
     *   - TWILIO_ACCOUNT_SID / TWILIO_AUTH_TOKEN / TWILIO_FROM_NUMBER are not set
     *   - The `to` phone number is null/undefined/empty
     *
     * Usage:
     *   await sendSms('+15551234567', 'Your step is overdue.');
     */
    import { logger } from '@/lib/logger';

    let twilioClient: import('twilio').Twilio | null = null;

    function getClient(): import('twilio').Twilio | null {
      const sid = process.env.TWILIO_ACCOUNT_SID;
      const token = process.env.TWILIO_AUTH_TOKEN;
      if (!sid || !token) return null;
      if (!twilioClient) {
        // Lazy-init: avoids crashing on module load when env vars are absent
        const twilio = require('twilio');
        twilioClient = twilio(sid, token) as import('twilio').Twilio;
      }
      return twilioClient;
    }

    export async function sendSms(to: string | null | undefined, body: string): Promise<void> {
      const fromNumber = process.env.TWILIO_FROM_NUMBER;

      if (!to) {
        logger.warn('[SMS] Skipping — recipient has no phoneNumber');
        return;
      }
      if (!fromNumber) {
        logger.warn('[SMS] Skipping — TWILIO_FROM_NUMBER not configured');
        return;
      }

      const client = getClient();
      if (!client) {
        logger.warn('[SMS] Skipping — TWILIO_ACCOUNT_SID or TWILIO_AUTH_TOKEN not configured');
        return;
      }

      try {
        await client.messages.create({ to, from: fromNumber, body });
        logger.info(`[SMS] Sent to ${to.slice(0, 6)}***`);
      } catch (err) {
        // Best-effort: log but never throw to callers
        logger.error('[SMS] Send failed', { error: err });
      }
    }
    ```
