### 1. Add PushToken table to Prisma schema
File: apps/web/prisma/schema.prisma (add model)

```prisma
model PushToken {
  id        String   @id @default(cuid())
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  token     String
  platform  String   // 'ios' | 'android'
  updatedAt DateTime @updatedAt
  createdAt DateTime @default(now())

  @@unique([userId, platform])  // one token per user per platform
  @@index([userId])
}
```

### 5. Mobile: register push token on login
File: apps/mobile/hooks/usePushNotifications.ts

```typescript
import * as Notifications from 'expo-notifications'
import { Platform } from 'react-native'
import { apiClient } from '@drivecommand/api-client'

export async function registerPushToken(authToken: string) {
  // Check/request permission
  const { status } = await Notifications.requestPermissionsAsync()
  if (status !== 'granted') return

  // Get Expo push token
  const tokenData = await Notifications.getExpoPushTokenAsync({
    projectId: 'your-eas-project-id',  // from app.json expo.extra.eas.projectId
  })

  // Register with backend
  await apiClient.registerPushToken(authToken, tokenData.data, Platform.OS as 'ios' | 'android')
}
```

Call `registerPushToken(token)` in the login flow (after successful login, non-blocking).

### 6. Mobile: handle notification taps
File: apps/mobile/app/_layout.tsx (add to root layout)

```typescript
import * as Notifications from 'expo-notifications'
import { useRouter } from 'expo-router'

// Set notification handler (show notification when app is foregrounded)
Notifications.setNotificationHandler({
  handleNotification: async () => ({
    shouldShowAlert: true,
    shouldPlaySound: true,
    shouldSetBadge: true,
  }),
})

// In root layout component:
useEffect(() => {
  const sub = Notifications.addNotificationResponseReceivedListener(response => {
    const data = response.notification.request.content.data
    if (data?.screen === 'messages') router.push('/(driver)/messages')
    if (data?.screen === 'loads') router.push('/(driver)/loads')
    if (data?.screen === 'documents') router.push('/(driver)/documents')
  })
  return () => sub.remove()
}, [])
```
