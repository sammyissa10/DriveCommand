**DriverHOSEntry model:**

```
model DriverHOSEntry {
  id        String        @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId  String        @db.Uuid
  driverId  String        @db.Uuid
  status    HOSDutyStatus
  startTime DateTime      @db.Timestamptz
  endTime   DateTime?     @db.Timestamptz
  notes     String?
  createdAt DateTime      @default(now()) @db.Timestamptz
  updatedAt DateTime      @updatedAt @db.Timestamptz

  tenant Tenant @relation(fields: [tenantId], references: [id])
  driver User   @relation(name: "DriverHOSEntries", fields: [driverId], references: [id])

  @@index([tenantId])
  @@index([driverId])
  @@index([driverId, startTime])
  @@index([tenantId, driverId, startTime])
}
```

**DriverIncident model:**

```
model DriverIncident {
  id          String           @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId    String           @db.Uuid
  driverId    String           @db.Uuid
  category    IncidentCategory
  severity    IncidentSeverity
  description String
  latitude    Decimal?         @db.Decimal(10, 8)
  longitude   Decimal?         @db.Decimal(11, 8)
  photoS3Key  String?
  reportedAt  DateTime         @db.Timestamptz
  notes       String?
  createdAt   DateTime         @default(now()) @db.Timestamptz
  updatedAt   DateTime         @updatedAt @db.Timestamptz

  tenant Tenant @relation(fields: [tenantId], references: [id])
  driver User   @relation(name: "DriverIncidents", fields: [driverId], references: [id])

  @@index([tenantId])
  @@index([driverId])
  @@index([tenantId, reportedAt])
  @@index([category])
  @@index([severity])
}
```
