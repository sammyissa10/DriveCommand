# Phase 29 — Plan 03: Shared Packages — Types, Validation, API Client

## Goal
Build the three shared packages that both apps/web and apps/mobile import. Extract existing TypeScript types and Zod schemas from the web app into framework-agnostic packages. Build the api-client package that the mobile app uses to call the existing Next.js API routes with Bearer token auth.

- Plan 29-01 complete (monorepo structure exists)
- Plan 29-02 complete (apps/mobile exists with package.json referencing workspace packages)

#### 3b. Create packages/api-client/src/client.ts
```typescript
import type { AuthSession, AuthUser } from '@drivecommand/types'

const getBaseUrl = () => {
  if (process.env.EXPO_PUBLIC_API_URL) return process.env.EXPO_PUBLIC_API_URL
  if (typeof window !== 'undefined') return ''
  return 'http://localhost:3000'
}

async function apiRequest<T>(
  path: string,
  options: RequestInit & { token?: string } = {}
): Promise<T> {
  const { token, ...fetchOptions } = options
  const url = `${getBaseUrl()}${path}`

  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
    ...(fetchOptions.headers as Record<string, string>),
  }

  if (token) {
    headers['Authorization'] = `Bearer ${token}`
  }

  const res = await fetch(url, { ...fetchOptions, headers })

  if (!res.ok) {
    const error = await res.json().catch(() => ({ error: 'Request failed' }))
    throw new Error(error.error || `HTTP ${res.status}`)
  }

  return res.json()
}

export const apiClient = {
  // Auth
  login: (email: string, password: string) =>
    apiRequest<AuthSession>('/api/auth/login', {
      method: 'POST',
      body: JSON.stringify({ email, password }),
    }),

  me: (token: string) =>
    apiRequest<AuthUser>('/api/auth/me', { token }),

  logout: (token: string) =>
    apiRequest<void>('/api/auth/logout', { method: 'POST', token }),

  // Push tokens
  registerPushToken: (token: string, pushToken: string, platform: 'ios' | 'android') =>
    apiRequest<void>('/api/push-tokens', {
      method: 'POST',
      token,
      body: JSON.stringify({ pushToken, platform }),
    }),

  // GPS
  reportGPS: (token: string, payload: {
    latitude: number
    longitude: number
    speed?: number
    heading?: number
    altitude?: number
    trackingToken: string
  }) =>
    apiRequest<void>('/api/gps/report', {
      method: 'POST',
      token,
      body: JSON.stringify(payload),
    }),

  getGPSLocations: (token: string) =>
    apiRequest<import('@drivecommand/types').GPSLocation[]>('/api/gps/locations', { token }),
}

export type { AuthSession, AuthUser }
```

#### 3c. Note on server actions
The mobile app calls REST API endpoints, not Next.js server actions directly. For endpoints that don't have a REST route yet (e.g., getMyLoads, getDrivers), Claude will create thin REST wrapper routes in apps/web/src/app/api/mobile/* during the relevant phase. This is done phase-by-phase, not all upfront.
