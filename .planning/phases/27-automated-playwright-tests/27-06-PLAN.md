        await page.goto('/carrier/reports/driver-pay');

        // Find the Mark as Paid button in any unpaid row
        const markPaidBtn = page.getByRole('button', { name: /Mark as Paid|Mark Paid/i }).first();

        await markPaidBtn.click();

        await page.waitForLoadState('networkidle');
        // The button should disappear or the row status should update to 'paid'
        const successIndicator = page.getByText(/Paid/i).first().or(
          page.locator('[data-status="paid"]').first()

      });
