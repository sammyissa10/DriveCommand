```ts
export interface AddressResult {
  formatted_address: string
  latitude: number
  longitude: number
  place_id: string // Nominatim's place_id as string
}
```

2. Create `apps/web/src/app/api/geocoding/autocomplete/route.ts`:
   - POST handler accepting JSON body `{ query: string }`
   - Validate: query must be string, length >= 3, max 300 chars. Return 400 if invalid.
   - Call Nominatim: `https://nominatim.openstreetmap.org/search` with params: `q=${query}`, `format=json`, `limit=6`, `countrycodes=us`, `addressdetails=1`
   - Set User-Agent header to `DriveCommand/1.0 (geocoding-proxy)` (Nominatim requires this)
   - Map response to `AddressResult[]`: format address from address parts (house_number + road, city/town/village, state + postcode) same as current web component does. Use Nominatim's `place_id` (convert to string), `lat` and `lon` parsed as floats.
   - Add response cache header: `Cache-Control: public, max-age=300` (5 min)
   - In-memory dedup: use a simple Map<string, {data, timestamp}> to cache results for 60 seconds. Key = lowercase trimmed query. Check cache before calling Nominatim.
   - No auth required (this is a public geocoding proxy for logged-in app users, but the route itself lives under /api so Next.js middleware can protect it if needed).
   - Return `NextResponse.json(results)` with status 200.
   - Wrap Nominatim fetch in try/catch, return empty array on failure (graceful degradation).
  </action>
